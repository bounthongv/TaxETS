-- Step 1: Create the database if it doesn't exist and select it for use.
CREATE DATABASE IF NOT EXISTS `taxets` DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE `taxets`;

-- Step 2: Create the main 'regimes' table.
DROP TABLE IF EXISTS `benchmark_profit_tax_regimes`;
CREATE TABLE `benchmark_profit_tax_regimes` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `name` VARCHAR(255) NOT NULL,
  `valid_from` DATE NOT NULL,
  `valid_to` DATE NULL,
  `standard_profit_tax_rate` DECIMAL(5, 4) NOT NULL,
  `micro_enterprise_turnover_threshold` BIGINT NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Step 3: Create the 'standard_sector_rates' table.
DROP TABLE IF EXISTS `benchmark_standard_sector_rates`;
CREATE TABLE `benchmark_standard_sector_rates` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `regime_id` INT NOT NULL,
  `sector_name` VARCHAR(255) NOT NULL,
  `profit_tax_rate` DECIMAL(5, 4) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`regime_id`) REFERENCES `benchmark_profit_tax_regimes`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Step 4: Create the 'mandatory_estimation_rates' table.
DROP TABLE IF EXISTS `benchmark_mandatory_estimation_rates`;
CREATE TABLE `benchmark_mandatory_estimation_rates` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `regime_id` INT NOT NULL,
  `activity_description` VARCHAR(255) NOT NULL,
  `profit_estimation_rate` DECIMAL(5, 4) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`regime_id`) REFERENCES `benchmark_profit_tax_regimes`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Step 5: Create the 'micro_enterprise_turnover_tax_rules' table.
DROP TABLE IF EXISTS `benchmark_micro_enterprise_turnover_tax_rules`;
CREATE TABLE `benchmark_micro_enterprise_turnover_tax_rules` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `regime_id` INT NOT NULL,
  `min_turnover` BIGINT NOT NULL DEFAULT 0,
  `max_turnover` BIGINT NOT NULL,
  `calculation_method` ENUM('RATE', 'LUMP_SUM', 'ZERO') NOT NULL,
  `production_rate` DECIMAL(5, 4) NULL,
  `commerce_rate` DECIMAL(5, 4) NULL,
  `services_rate` DECIMAL(5, 4) NULL,
  `lump_sum_description` TEXT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`regime_id`) REFERENCES `benchmark_profit_tax_regimes`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
