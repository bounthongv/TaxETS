-- ===================================================================================
-- SCRIPT TO CREATE TE REPOSITORY TABLES AND INSERT PROFIT TAX DATA
-- ===================================================================================

-- Step 1: Create the `te_provisions` master table.
DROP TABLE IF EXISTS `te_provisions`;
CREATE TABLE `te_provisions` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `effective_from_date` DATE NOT NULL,
  `effective_to_date` DATE NULL,
  `legal_basis` VARCHAR(255) NULL,
  `tax_type` VARCHAR(50) NOT NULL,
  `te_type` ENUM('Rate Relief', 'Exemption', 'Deduction', 'Deferral') NOT NULL,
  `description` TEXT NULL,
  `purpose_objective` VARCHAR(255) NULL,
  `priority` INT NOT NULL DEFAULT 100,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Step 2: Create the `te_conditions` table.
DROP TABLE IF EXISTS `te_conditions`;
CREATE TABLE `te_conditions` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `te_provision_id` INT NOT NULL,
  `data_field_name` VARCHAR(100) NOT NULL,
  `operator` ENUM('=', '!=', '>', '>=', '<', '<=', 'IN', 'BETWEEN') NOT NULL,
  `value` VARCHAR(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`te_provision_id`) REFERENCES `te_provisions`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Step 3: Create the `te_effects` table.
DROP TABLE IF EXISTS `te_effects`;
CREATE TABLE `te_effects` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `te_provision_id` INT NOT NULL,
  `effect_type` ENUM('SET_TAX_RATE', 'REDUCE_TAXABLE_BASE', 'EXEMPTION') NOT NULL,
  `parameter_name` VARCHAR(100) NULL,
  `value` VARCHAR(255) NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`te_provision_id`) REFERENCES `te_provisions`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- This index allows the system to quickly fetch the effect(s) for a specific TE provision
-- after it has been identified as applicable.
